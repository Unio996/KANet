## 功能需求

### 5b.1 exchange-machine maker auto-pay-give 分支 (exchange-machine.js)

Context: Seeder (maker) 发 BUY offer 给 USDT 收 KAS. seller (taker) 接单后付 KAS 到 user_kasia, 验证过 → seeder 要**自动发 USDT 给 seller**.

当前 exchange-machine 在 delivered → completed 转换时:
- taker auto-pay 分支: taker 付 USDT 给 maker (现有)
- **maker auto-pay-give 分支**: maker 发 give 资产 (USDT) 给 taker (**新加**)

判定条件:
- offer.maker_relay_id 对应 seeder relay (查 agent_wallets 或 relay_nodes)
- offer.give.asset === 'USDT'
- offer.give.chain 是 EVM 链
- 对应 retail_dex_buy_publications 行 state='filled'

执行逻辑:
```js
// 在 handleDelivered 或 transition completed 分支里加:
if (isSeederBuyMakerOffer(offer)) {
  // 1. 从 retail_dex_buy_publications 查对应行
  const pub = sqlite.prepare("SELECT * FROM retail_dex_buy_publications WHERE seeder_publish_offer_id = ?").get(offer.id);
  if (!pub) throw new Error(`no_publication_for_offer_${offer.id.slice(0,8)}`);

  // 2. 查 seeder 私钥 + taker 收 USDT 地址
  const privkey = await _loadSeederPrivkey(offer.maker_relay_id, pub.pay_chain);
  const takerAddr = offer.taker_payment_address;  // seller 的 EVM 收款地址

  // 3. 调 transferUsdt 发 USDT 给 seller
  const { txHash, error } = await transferUsdt({
    chain: pub.pay_chain,  // bnb / eth / ...
    fromPrivkey: privkey,
    toAddress: takerAddr,
    amountUsdt: pub.total_usdt,
  });
  if (error) {
    sqlite.prepare("UPDATE retail_dex_buy_publications SET state='failed', error_reason=? WHERE id=?").run(error, pub.id);
    throw new Error(`maker_auto_pay_failed: ${error}`);
  }

  // 4. 落库 + 推进 state
  sqlite.prepare("UPDATE retail_dex_buy_publications SET state='completed', filled_at=datetime('now'), usdt_refund_tx=NULL WHERE id=?").run(pub.id);
}
```

注意:
- 不能 mock 掉 transferUsdt (真 EVM 广播, 不过 smoke 里用 _testInjectTransferUsdt 注入)
- 只在 pub.state='filled' 时推进 (避免 double pay)
- failed 要落 error_reason (供 UI 诊断)

### 5b.2 超时退款 worker (market-seeder.js)

新 export:
```js
export function startSeederRefundWorker() { setInterval(refundWorkerTick, 30_000); }
export function stopSeederRefundWorker() { clearInterval(...); }
export async function refundWorkerTick() {
  // 1. 扫 state='published' AND expires_at < now
  const expired = sqlite.prepare(`
    SELECT * FROM retail_dex_buy_publications
    WHERE state = 'published' AND expires_at < datetime('now')
  `).all();

  for (const pub of expired) {
    try {
      // 2. 查 offer 状态 — 若 open (未被 accept) → 退
      const offer = sqlite.prepare("SELECT * FROM exchange_offers WHERE id = ?").get(pub.seeder_publish_offer_id);
      if (!offer || offer.state !== 'open') continue;  // 已 match 交给 exchange-machine 处理

      // 3. state → refunding (不可回, 防竞态)
      sqlite.prepare("UPDATE retail_dex_buy_publications SET state='refunding', updated_at=datetime('now') WHERE id=?").run(pub.id);

      // 4. cancel_v1 广播 (释放 exchange_offers fund_lock)
      const cancelRes = await sendCommandAsync(pub.seeder_relay_id, {
        type: 'send_broadcast',
        payload: { t: 'kanet_exchange_cancel_v1', offer_id: pub.seeder_publish_offer_id, reason: 'timeout' },
      });
      if (cancelRes.error) throw new Error(`cancel_broadcast_failed: ${cancelRes.error}`);

      // 5. transferUsdt 退 USDT 回用户 (user 收 USDT 地址查 retail_dex_orders.pay_address)
      const order = sqlite.prepare("SELECT pay_address, pay_chain FROM retail_dex_orders WHERE id IN (SELECT order_id FROM retail_dex_buy_publications WHERE id = ?)").get(pub.id);
      const { txHash, error } = await transferUsdt({
        chain: order?.pay_chain || pub.pay_chain,
        fromPrivkey: await _loadSeederPrivkey(pub.seeder_relay_id, pub.pay_chain),
        toAddress: order?.pay_address,
        amountUsdt: pub.total_usdt,
      });
      if (error) throw new Error(`refund_transfer_failed: ${error}`);

      // 6. state → refunded
      sqlite.prepare("UPDATE retail_dex_buy_publications SET state='refunded', usdt_refund_tx=?, updated_at=datetime('now') WHERE id=?").run(txHash, pub.id);
    } catch (err) {
      // 失败 → failed + error_reason (可人工介入)
      sqlite.prepare("UPDATE retail_dex_buy_publications SET state='failed', error_reason=? WHERE id=?").run(err.message, pub.id);
    }
  }
}
```

index.js 里 startup 拉起 startSeederRefundWorker().
