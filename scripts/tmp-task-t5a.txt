### 5a.3 Seeder USDT 入账 scanner (新)

新独立函数 startSeederDepositWatcher(),放在 market-seeder.js 里, 和 startMarketSeeder 平行 export:

```js
export function startSeederDepositWatcher() {
  // 每 30s tick 一次
  setInterval(async () => {
    // 1. 查 retail_dex_buy_publications WHERE state='awaiting_deposit'
    // 2. 对每行, 调 getTokenBalance('bnb', seeder_bsc_addr, 'usdt')
    //    算 current - baseline >= expected_usdt (1% tolerance)
    // 3. 若到账 → state='deposited' + deposited_at=now
    // 4. 下一 tick: state='deposited' 的行 → 调 publishSeedOrder buy, 成功后 state='published'
  }, 30000);
}
```

要在 index.js startup 里调 startSeederDepositWatcher() (一起 kanet-start 拉起).

### 5a.4 retail-dex.js 触发

在 createOrder 成功后, 若 order_type='limit' (M2 限价) + side='buy_kas' → 写 retail_dex_buy_publications 行 state='awaiting_deposit':

```js
async function _triggerBuyPublication({ orderId, userAddr, qty, price, brokerRelayId }) {
  const expected_usdt = (parseFloat(qty) * parseFloat(price)).toFixed(6);
  // seeder_bsc_addr 从 agent_wallets 查 relay_node_id=brokerRelayId AND chain='bnb' AND is_default=1
  const seeder_bsc_addr = sqlite.prepare(
    "SELECT address FROM agent_wallets WHERE relay_node_id = ? AND chain = 'bnb' AND is_default = 1 LIMIT 1"
  ).get(brokerRelayId)?.address;
  if (!seeder_bsc_addr) throw new Error('seeder_bsc_addr_missing');
  sqlite.prepare(`
    INSERT INTO retail_dex_buy_publications (
      id, order_id, user_kasia_address, qty_kas, price_usdt_per_kas, expected_usdt,
      seeder_bsc_address, state, created_at
    ) VALUES (?, ?, ?, ?, ?, ?, ?, 'awaiting_deposit', ?)
  `).run(randomUUID(), orderId, userAddr, qty, price, expected_usdt, seeder_bsc_addr, new Date().toISOString());
}
```
