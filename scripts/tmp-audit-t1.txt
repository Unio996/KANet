## EVM 实现骨架
```js
async function _evmBalance(chain, address, tokenSymbol) {
  const { ethers } = await import('ethers');
  const rpcUrl = EVM_RPC[chain];
  const token = EVM_TOKENS[chain]?.[tokenSymbol];
  if (!token) return { balance: 0, error: `no_${tokenSymbol}_on_${chain}` };
  const provider = new ethers.JsonRpcProvider(rpcUrl);
  try {
    const erc20 = new ethers.Contract(token.address, ['function balanceOf(address) view returns (uint256)'], provider);
    const raw = await erc20.balanceOf(address);
    const balance = parseFloat(ethers.formatUnits(raw, token.decimals));
    return { balance, decimals: token.decimals, tokenAddress: token.address };
  } catch (err) {
    return { balance: 0, error: err.message };
  } finally {
    try { provider.destroy?.(); } catch {}
  }
}
```
