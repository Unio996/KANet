  _source: { onchainTxid: on.verboseData.transactionId, node: 'kaspad 2.0.1 simnet PID15972 (sha256 8afe6a68...)', nodeStorageMass: on.storageMass ?? null, nodeMass: on.mass ?? null, nodeComputeMass: on.verboseData.computeMass ?? null },
  params: {
    marketId: g.marketId, minBet: g.minBet, deadlineMs: g.deadlineMs, sealCount: g.sealCount,
    committeePubkeyHex: g.committeePubkeyHex, rootCloseTmplHash: g.rootCloseTmplHash, shardLeafOwnRedeemLen: g.shardLeafOwnRedeemLen,
    leafCovId: g.shardLeafCovId,
    stateAtSeal: bak.bet2.newState,
    leafOutpoint: { txid: on.inputs[0].previousOutpoint.transactionId, vout: on.inputs[0].previousOutpoint.index },
    heldOutpoint: { txid: on.inputs[1].previousOutpoint.transactionId, vout: on.inputs[1].previousOutpoint.index },
    feeOutpoint: { txid: on.inputs[2].previousOutpoint.transactionId, vout: on.inputs[2].previousOutpoint.index },
    feeValueSompi: String(BigInt(on.outputs[2].value) + BigInt(bak.marketSeal.requiredFee)),
    relaySpkScriptHex: spk(on.outputs[2]),
  },
  expected: {
    version: on.version, lockTime: on.lockTime,
    inputs: on.inputs.map((i) => ({ txid: i.previousOutpoint.transactionId, vout: i.previousOutpoint.index, sequence: i.sequence, sigOpCount: i.sigOpCount, computeBudget: i.computeBudget, signatureScriptHex: i.signatureScript })),
    outputs: on.outputs.map((o) => ({ value: o.value, spkScriptHex: spk(o), covenantId: o.covenant?.covenantId ?? null, authorizingInput: o.covenant?.authorizingInput ?? null })),
    unsignedTxidNoWitness: on.verboseData.transactionId,
  },
};

fx.expected.inputs[2].signatureScriptHex = null;
