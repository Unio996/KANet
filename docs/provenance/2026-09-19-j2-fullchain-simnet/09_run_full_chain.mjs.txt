const jsonSafe = (k, v) => (typeof v === 'bigint' ? v.toString() : v);
async function getUtxos(addr) {
  const res = await rpc.getUtxosByAddresses({ addresses: [addr] });
  return (res.entries || []).map((e) => ({ txid: e.outpoint.transactionId, vout: e.outpoint.index, value: BigInt(e.amount), scriptPublicKeyHex: '0x' + e.scriptPublicKey.script, daaScore: BigInt(e.blockDaaScore ?? 0) }));
}
async function mineOne(payAddress) {
  const { block } = await rpc.getBlockTemplate({ payAddress });
  return rpc.submitBlock({ block, allowNonDAABlocks: true });
}
async function pmtSnapshot(label) {
  const d = await rpc.getBlockDagInfo();
  const s = { label, pastMedianTimeMs: String(d.pastMedianTime), nowMs: Date.now(), lagSeconds: (Date.now() - Number(d.pastMedianTime)) / 1000, virtualDaaScore: String(d.virtualDaaScore) };
  console.log(`[pmt] ${label}: pastMedianTime=${s.pastMedianTimeMs} lag=${s.lagSeconds.toFixed(1)}s daa=${s.virtualDaaScore}`);
  return s;
}
/** 提交前的节点权威值: getMempoolEntry原始返回(去掉超长sigScript, 只留长度+sha256前16位)。 */
async function mempoolRaw(txid) {
  try {
    const entry = await rpc.getMempoolEntry({ transactionId: txid, includeOrphanPool: true, filterTransactionPool: false });
    const j = JSON.parse(JSON.stringify(entry, jsonSafe));
    const me = j.mempoolEntry ?? j.entry ?? j;
    const tx = me.transaction ?? {};
    if (Array.isArray(tx.inputs)) tx.inputs = tx.inputs.map((i) => ({ ...i, signatureScript: undefined, signatureScriptBytes: String(i.signatureScript ?? '').length / 2, signatureScriptSha256_16: createHash('sha256').update(Buffer.from(String(i.signatureScript ?? ''), 'hex')).digest('hex').slice(0, 16) }));
    return { storageMass: String(me.mass ?? tx.mass ?? ''), computeMass: String(tx.verboseData?.computeMass ?? me.verboseData?.computeMass ?? ''), raw: j };
  } catch (e) { return { error: e.message }; }
}
async function submitAndConfirm(tx, label, minerAddr, signals) {
  console.log(`\n--- 提交 ${label}: txid=${tx.id} version=${tx.version} 断言信号=${JSON.stringify(signals)} 本地wasm(诊断)=${(() => { try { return kaspa.calculateTransactionMass(NETWORK, tx); } catch { return 'n/a'; } })()} ---`);
  let res;
  try { res = await rpc.submitTransaction({ transaction: tx, allowOrphan: false }); } catch (e) { console.log(`${label} submitTransaction被拒: ${e.message}`); throw e; }
  console.log(`${label} submit result:`, JSON.stringify(res, jsonSafe));
  const mp = await mempoolRaw(tx.id);
  console.log(`${label} 节点getMempoolEntry: storageMass=${mp.storageMass} computeMass=${mp.computeMass}${mp.error ? ' ERR=' + mp.error : ''}`);
  await mineOne(minerAddr);
  console.log(`已挖1块确认 ${label}`);
  // Bettor纪律: 断言信号与节点值不等即停, 不带着不等继续往下跑(回头修公式)。
  if (String(signals.storageMass) !== String(mp.storageMass) || String(signals.computeMass) !== String(mp.computeMass)) {
    chain[`STOP_mismatch_${label}`] = { txid: tx.id, assertion: signals, node: { storageMass: mp.storageMass, computeMass: mp.computeMass } };
    saveChain();
    throw new Error(`STOP: ${label} 断言信号(storage=${signals.storageMass}, compute=${signals.computeMass}) != 节点getMempoolEntry(storage=${mp.storageMass}, compute=${mp.computeMass}) — 停下, 回头修公式`);
  }
  console.log(`  ✓ ${label}: 断言信号 == 节点值 (storage=${mp.storageMass}, compute=${mp.computeMass})`);
  return { mp, signals };
}

async function pickFeeUtxoCandidates(maxValue = 100_000_000n) {
  const utxos = await getUtxos(relayAddr);
  return utxos.filter((u) => u.value <= maxValue).sort((a, b) => (a.value < b.value ? 1 : -1));
}
async function tryWithFeeUtxoRetry(buildOnce) {
  const candidates = await pickFeeUtxoCandidates();
  if (!candidates.length) throw new Error('没有<=1.0KAS的候选fee UTXO了');
  let lastErr;
  for (const feeUtxo of candidates) {
    try { return await buildOnce(feeUtxo); } catch (e) {
      if (!/assertMassWithinCeiling|no_viable_change_shape/.test(e.message)) throw e;
      console.log(`  [换面值重试] fee=${feeUtxo.value} 被拒: ${e.message.slice(0, 160)}`);
      lastErr = e;
    }
  }
  throw new Error(`tryWithFeeUtxoRetry: 全部${candidates.length}个候选都不可行, 最后一次: ${lastErr.message}`);
}
function signAndCheck(built, label) {
  const tx = kaspa.Transaction.deserializeFromSafeJSON(built.txJson);
  signOnlyDeclaredInputs({ tx, signInputIndices: built.signInputIndices, privateKey: relayPriv, kaspa });
  tx.finalize();
  const idCheck = assertFinalTxid(tx, built.expectedTxid);
  if (!idCheck.ok) throw new Error(`${label}签名后txid不一致: ${idCheck.actualTxid} != ${built.expectedTxid}`);
  return tx;
}

  const mergedArtifact = computeKttGenesisArtifact({ amount: newState.pool_value, ownerCovIdHex: chain.genesis.shardLeafCovId });
  const ticketCompiled = compileSilV100(TICKET_PATH, [ctorBytes32V100(bettorPk), ctorIntV100(side), ctorIntV100(stake), ctorBytes32V100(chain.genesis.marketId)], 'PoolSideTicket');
  const tArt = extractTemplateArtifactV100(ticketCompiled);

    console.log(`${stepKey} 尝试fee输入面值:`, feeUtxo.value.toString());

      kaspa, network: NETWORK, leafRedeemScript: leafRedeem.script, leafStateLayout: leafRedeem.stateLayout,
      leafOutpoint, leafCovId: chain.genesis.shardLeafCovId, currentState, newState, heldInput,

  const closedState = { ...chain.bet2.newState, closed: 1, winningSide: chain.closeCommit.newWinningSide, payoutRoot: chain.closeCommit.newPayoutRoot };
  const { tx, built, signals } = await tryWithFeeUtxoRetry(async (feeUtxo) => {
    console.log('convert_to_claim 尝试fee输入面值:', feeUtxo.value.toString());
    const built = buildConvertToClaimTxJson({
      kaspa, network: NETWORK, marketId: chain.genesis.marketId, committeePubkeyHex: chain.genesis.committeePubkeyHex, deadlineMs: chain.genesis.deadlineMs, rootCloseTmplHash: chain.genesis.rootCloseTmplHash,
      rootCloseOutpoint: { txid: chain.closeCommit.txid, vout: CLOSE_COMMIT_ROOTCLOSE_OUT_INDEX }, rootCloseCovId: chain.marketSeal.rootCloseCovId, closedState,
      heldTokenOutpoint: { txid: chain.marketSeal.txid, vout: MARKET_SEAL_TOKEN_OUT_INDEX },
      tokPrefixHex, tokSuffixHex, feeUtxo, relayChangeScriptPublicKeyHex: relaySpkHex, absFeeCapSompi: loadFeeProfileCap('convert_to_claim'),
    });
    const tx = signAndCheck(built, 'convert_to_claim');
    return { tx, built, signals: signalsFor(tx, [true, true, false]) };
  });
  const r = await submitAndConfirm(tx, 'convert_to_claim', minerAddr, signals);
  chain.convertToClaim = { txid: tx.id, claimCovId: built.claimCovId, tokenCovId: built.tokenCovId, node: r.mp, assertion: r.signals, requiredFee: String(built.requiredFee), claimOutpoint: { txid: tx.id, vout: CONVERT_TO_CLAIM_CLAIM_OUT_INDEX }, tokenOutpoint: { txid: tx.id, vout: CONVERT_TO_CLAIM_TOKEN_OUT_INDEX } };
  saveChain(); console.log('=== 步骤6(convert_to_claim)完成 txid=', tx.id);
} else console.log('=== 步骤6已完成, 跳过 ===', chain.convertToClaim.txid);

console.log('\n========== 全部完成, 各步 断言信号 vs 节点getMempoolEntry值 ==========');
for (const k of ['genesis', 'bet1', 'bet2', 'marketSeal', 'closeCommit', 'convertToClaim']) {
  const s = chain[k]; if (!s) continue;
  console.log(`${k.padEnd(15)} txid=${s.txid.slice(0, 12)}…  断言 storage=${s.assertion?.storageMass} compute=${s.assertion?.computeMass}  |  节点 storage=${s.node?.storageMass} compute=${s.node?.computeMass}`);
}
await rpc.disconnect();
process.exit(0);
