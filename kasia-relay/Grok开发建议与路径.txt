```ts
// poller.ts (核心 <100 行)
class KasiaPoller extends EventEmitter {
  private backoff = { min: 500, max: 5000, current: 500, multiplier: 1.5 };
  private fetchTimer: NodeJS.Timeout | null = null;
  private lastMessageTs = 0;

  async start() {
    // TriggerWatcher
    this.watcher.subscribeUtxosChanged([myAddress]);
    this.watcher.on('utxosChanged', () => this.scheduleFetch(true)); // 立即

    this.watcher.subscribeNewBlock();
    this.watcher.on('newBlock', () => this.scheduleFetch(false));

    setInterval(() => this.scheduleFetch(false), 800); // 心跳
  }

  private scheduleFetch(immediate: boolean) {
    if (this.fetchTimer) clearTimeout(this.fetchTimer);
    const delay = immediate ? 0 : this.backoff.current;
    this.fetchTimer = setTimeout(() => this.fetchMessages(), delay);
  }

  private async fetchMessages() {
    const now = Date.now();
    let hasNew = false;

    // ... (cursor + kasia_get_conversations + kasia_get_messages 逻辑同之前)

    for (const msg of newMsgs) {
      if (this.dedup.isDuplicate(msg)) continue;
      this.dedup.add(msg);
      this.outbox.markInbound(msg); // 可选
      this.router.route(msg); // 推到 Queue → Router → Gateway
      hasNew = true;
    }

    // 自适应核心
    if (hasNew) {
      this.backoff.current = this.backoff.min; // 热状态
      this.lastMessageTs = now;
    } else if (now - this.lastMessageTs > 30000) { // 空闲 30s 后才退避
      this.backoff.current = Math.min(this.backoff.current * this.backoff.multiplier, this.backoff.max);
    }
  }
}
```
